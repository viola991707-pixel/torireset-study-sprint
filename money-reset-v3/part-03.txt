  // months and interest to clear a balance paying `pay` a month (null when the payment never catches up with interest)
  function payoffSim(bal, rate, pay) {
    if (!(bal > 0)) return { months: 0, interest: 0 };
    if (!(pay > 0)) return null;
    const r = (rate || 0) / 1200;
    if (r > 0 && pay <= bal * r) return null;
    let months = 0, interest = 0;
    while (bal > 0.005 && months < 1200) {
      const i = bal * r; interest += i; bal = bal + i - pay; months++;
    }
    return { months, interest: round2(interest) };
  }
