  /* ---------- backup ---------- */
  function download(name, text, mime) {
    try {
      const url = URL.createObjectURL(new Blob([text], { type: mime || 'application/json' }));
      const a = document.createElement('a'); a.href = url; a.download = name; document.body.appendChild(a); a.click();
      setTimeout(() => { URL.revokeObjectURL(url); a.remove(); }, 500); return true;
    } catch (e) { return false; }
  }
  function backupText() { return JSON.stringify(S); }
  function doBackup() {
    const ok = download('money-reset-backup-' + today() + '.json', backupText());
    if (ok) { S.lastBackup = today(); S.changes = 0; save(false); render(); toast('Backup downloaded. Keep it somewhere safe.'); }
    else toast('Your browser blocked the download. Use “Copy as text” in More → Backup instead.');
  }
  function restoreFrom(text) {
    let data; try { data = C.validateAppData(JSON.parse(text)); } catch (e) { toast(e && e.message && e.message.startsWith('This file') ? e.message : 'That does not look like a Money Reset backup.'); return; }
    confirmModal('Replace what is on this device?', 'This backup has ' + data.profiles.length + (data.profiles.length === 1 ? ' budget' : ' budgets') + ' (' + data.profiles.map((p) => esc(p.name)).join(', ') + '). It will replace the numbers currently on this device.', 'Restore', () => {
      S = data; P = active(); S.changes = 0; sub = null; view = 'today'; save(false); render(); toast('Restored.');
    });
  }
  function csv() {
    // a leading = + - @ would be run as a formula by spreadsheets, so such text gets a quote mark in front
    const q = (v) => { let s = String(v == null ? '' : v); if (/^[=+\-@\t\r]/.test(s)) s = "'" + s; return '"' + s.replace(/"/g, '""') + '"'; };
    const d = stmtData();
    const rows = ['Date,Type,Amount,Category,Note'].concat(d.rows.slice().reverse().map((x) => [x.e.date, x.e.type, x.e.delta || 0, q(x.e.cat), q(x.e.type === 'spend' ? x.e.note : entryText(x.e))].join(',')));
    // the BOM makes Excel read the file as UTF-8, so Cyrillic and accents are not garbled
    if (!download('statement-' + d.from + '_' + d.to + '.csv', '﻿' + rows.join('\r\n'), 'text/csv;charset=utf-8')) toast('Your browser blocked the download.');
  }
