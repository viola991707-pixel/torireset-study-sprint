  function money(fd, name, o) {
    o = o || {}; const v = C.parseMoney(fd.get(name));
    if (!Number.isFinite(v)) return o.optional && !String(fd.get(name) || '').trim() ? null : 'Type an amount, like 12.50';
    if (!o.negOk && v <= 0) return 'The amount needs to be more than zero.';
    if (Math.abs(v) > 1e9) return 'That is a very big number. Please check it.';
    return v;
  }

    const edit = !!d;

    modal(edit ? 'Edit debt' : 'Add a debt',

      fld('Due day of the month (optional)', 'day', edit && lbill ? lbill.day : '', { num: true, hint: edit && lbill ? 'The minimum bill follows what you type here.' : 'Adds the minimum to your Bills.' }),
      {

        extra: edit ? '<button type="button" class="danger" data-act="debt-del" data-id="' + d.id + '">Delete</button>' : '',
        onSubmit(fd) {
          const name = String(fd.get('name') || '').trim(); if (!name) return 'Give it a name.';

          const min = money(fd, 'min', { optional: true }); if (isErr(min)) return min;

          const day = parseInt(fd.get('day'), 10); const dayOk = day >= 1 && day <= 31;
          if (edit) {
            d.balance0 = C.round2(bal + (d.balance0 - C.debtBalance(P, d))); const oldName = d.name; d.name = name; d.min = min || 0; d.rate = rate || 0;
            // the minimum bill follows the debt, so the plan never counts an old amount
            if (lbill && d.min) { lbill.amount = d.min; if (dayOk) lbill.day = day; if (lbill.name === oldName + ' minimum') lbill.name = name + ' minimum'; }
            else if (!lbill && d.min && dayOk) P.bills.push({ id: C.uid(), name: name + ' minimum', amount: d.min, freq: 'monthly', day, debtId: d.id, start: today() });
            save(); render();
            if (lbill && !d.min) {
              const bid = lbill.id;
              setTimeout(() => confirmModal('Remove the minimum bill too?', 'You cleared the minimum payment, but &ldquo;' + esc(lbill.name) + '&rdquo; is still in your Bills (' + esc(fmt(lbill.amount)) + ').', 'Remove it', () => { P.bills = P.bills.filter((b) => b.id !== bid); save(); render(); toast('Bill removed'); }), 0);
            }
            return;
          }
          const nd = { id: C.uid(), name, balance0: bal, min: min || 0, rate: rate || 0 }; P.debts.push(nd);
          if (nd.min && dayOk) P.bills.push({ id: C.uid(), name: name + ' minimum', amount: nd.min, freq: 'monthly', day, debtId: nd.id, start: today() });
          save(); render();
        },
      });
