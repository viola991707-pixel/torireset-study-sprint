  function syncFromOtherTab() {
    let raw = null; try { raw = localStorage.getItem(KEY); } catch (e) { return; }
    if (!raw) return;
    try { S = C.validateAppData(JSON.parse(raw)); P = active(); ses.syncPending = false; render(); toast('Updated from another tab'); } catch (e) { /* keep what this tab has */ }
  }
