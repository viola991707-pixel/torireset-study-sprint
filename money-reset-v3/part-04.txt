  function readSched(fd, withEarly) {
    const freq = fd.get('freq'); const it = { freq };
    if (freq === 'monthly' || freq === 'twice') {
      const d = parseInt(fd.get('day'), 10);
      if (!(d >= 1 && d <= 31)) return 'Pick a day between 1 and 31.';
      it.day = d;
      if (freq === 'twice') {
        const d2 = parseInt(fd.get('day2'), 10); if (!(d2 >= 1 && d2 <= 31) || d2 === d) return 'Pick two different days.';
        if (d >= 29 && d2 >= 29) return 'Days 29 to 31 do not exist in every month, so two of them would merge in February. Pick one day up to the 28th.';
        it.day2 = d2;
      }
    } else if (freq === 'biweekly' || freq === 'weekly' || freq === 'months') {
      const a = fd.get('anchor'); if (!isDate(a)) return 'Pick a date.'; it.anchor = a;
      if (freq === 'months') it.every = parseInt(fd.get('every'), 10) || 6;
    } else if (freq === 'once') {
      const d = fd.get('date'); if (!isDate(d)) return 'Pick a date.'; it.date = d;
    }
    if (withEarly) it.early = !!fd.get('early');
    return it;
  }
